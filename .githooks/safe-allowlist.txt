# Allowlist for the pre-push hook — suppresses known false positives.
#
# One regex per line. Blank lines and '# ...' comments are ignored.
# Matching is case-insensitive against the entire hit line
# (the line from `grep -In`, e.g. "42:+some code here" or
# "3:[commit-msg] abc1234 subject line").
#
# If ANY pattern here matches the hit line, the hit is dropped.
#
# Prefer specific patterns over broad ones — an over-broad allowlist
# defeats the point of the hook. Anchor on file/path or unique context
# where possible.
#
# Examples:
#   # ignore a client-name keyword mentioned only in the example template
#   sensitive-keywords\.example
#
#   # allow the word "session" (which contains "ses") in test files
#   tests/.*session
#
#   # allow a specific known-safe token literal used in fixtures
#   fixtures/.*dummy-token-abc123
example\.com
# CSS comment separator banners (runs of '=') trip the AWS-secret-key heuristic
={10,}
# the crawler User-Agent embeds the project's own repo URL
github\.com/hashview/hashview
# inline SVG data: URIs reference the standard SVG namespace URL
w3\.org/2000/svg
# open-redirect guard comment/tests use the reserved evil.example host
evil\.example
# "regardless" contains the 'gard' client keyword as a substring
regardless

# "gard" keyword substring inside the ordinary English word "regardless"
# (CHANGELOG + a code comment) — not a credential.
regardless

# Test-only dummy credential literals in the API test fixtures
# (tests/unit/test_api_customer_hashfiles.py admin_user fixture).
password="hashed-pw"
api_key="user-api-key-admin"
src_ip="127\.0\.0\.1"
